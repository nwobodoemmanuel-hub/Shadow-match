let puzzles = [], currentRound = 0, score = 0, streak = 0, bestStreak = 0;
let answered = false, timerInterval = null, timeLeft = 0;
const TOTAL_ROUNDS = 12;
const TIME_LIMITS = { easy: 20, medium: 15, hard: 10 };
const POINTS = { easy: 100, medium: 150, hard: 220 };
const STREAK_BONUS = 50;

function shuffle(arr) {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

function init() {
  currentRound = 0; score = 0; streak = 0; bestStreak = 0;
  answered = false;
  puzzles = shuffle(ALL_PUZZLES).slice(0, TOTAL_ROUNDS);

  document.getElementById('endScreen').classList.remove('show');
  document.getElementById('endScreen').style.display = 'none';
  document.getElementById('gameContent').style.display = '';

  updateHUD();
  renderRound();
}

function updateHUD() {
  document.getElementById('scoreDisplay').textContent = score;
  document.getElementById('streakDisplay').textContent = streak;
}

function renderRound() {
  answered = false;
  stopTimer();

  const puzzle = puzzles[currentRound];
  document.getElementById('roundTag').textContent = `Round ${currentRound + 1} / ${TOTAL_ROUNDS}`;

  const diffEl = document.getElementById('diffTag');
  diffEl.textContent = puzzle.difficulty.charAt(0).toUpperCase() + puzzle.difficulty.slice(1);
  diffEl.className = 'difficulty-tag diff-' + puzzle.difficulty;

  document.getElementById('progressFill').style.width = (currentRound / TOTAL_ROUNDS * 100) + '%';
  document.getElementById('objectSvg').innerHTML = puzzle.objectSvg;
  document.getElementById('objNameLabel').textContent = puzzle.label;

  // Hints
  const hintLines = document.getElementById('hintLines');
  hintLines.innerHTML = puzzle.hints.map(h =>
    `<div class="hint-clue"><div class="clue-dot"></div><div class="clue-text">${h}</div></div>`
  ).join('');

  // Build shuffled options
  const indices = [0, 1, 2, 3];
  const shuffledIdx = shuffle(indices);
  const grid = document.getElementById('shadowsGrid');
  grid.innerHTML = '';

  shuffledIdx.forEach((origIdx, pos) => {
    const btn = document.createElement('div');
    btn.className = 'shadow-option';
    btn.innerHTML = `
      <span class="opt-num">${pos + 1}</span>
      <svg viewBox="0 0 100 100">${puzzle.shadows[origIdx]}</svg>
    `;
    btn.addEventListener('click', () => handleAnswer(btn, origIdx === puzzle.correctIdx, puzzle));
    grid.appendChild(btn);
  });

  // Feedback + next
  const fb = document.getElementById('feedbackBar');
  fb.className = 'feedback';
  fb.textContent = '';
  document.getElementById('nextBtn').className = 'next-btn';

  startTimer(TIME_LIMITS[puzzle.difficulty], puzzle);
}

function startTimer(seconds, puzzle) {
  timeLeft = seconds;
  const ring = document.getElementById('timerRing');
  const num = document.getElementById('timerNum');
  const circumference = 175.9;

  ring.style.stroke = 'var(--accent)';
  ring.style.strokeDashoffset = '0';
  num.textContent = seconds;

  timerInterval = setInterval(() => {
    timeLeft--;
    const progress = timeLeft / seconds;
    ring.style.strokeDashoffset = circumference * (1 - progress);
    num.textContent = timeLeft;

    if (progress < 0.33) ring.style.stroke = 'var(--danger)';
    else if (progress < 0.6) ring.style.stroke = 'var(--warn)';

    if (timeLeft <= 0) {
      clearInterval(timerInterval);
      handleTimeout(puzzle);
    }
  }, 1000);
}

function stopTimer() {
  clearInterval(timerInterval);
}

function handleTimeout(puzzle) {
  if (answered) return;
  answered = true;
  streak = 0;
  updateHUD();

  const allBtns = document.querySelectorAll('.shadow-option');
  allBtns.forEach(b => {
    b.classList.add('disabled', 'dimmed');
    const svg = b.querySelector('svg');
  });
  // Reveal correct
  allBtns.forEach((b, i) => {
    const svg = b.querySelector('svg');
    const origIdx = getOrigIdxFromBtn(b, puzzle);
    if (origIdx === puzzle.correctIdx) {
      b.classList.remove('dimmed');
      b.classList.add('correct');
      b.innerHTML += `<div class="result-badge badge-correct">✓</div>`;
    }
  });

  const fb = document.getElementById('feedbackBar');
  fb.className = 'feedback timeout show';
  fb.textContent = `⏱ Time's up! The correct shadow is highlighted above.`;

  showNextBtn();
}

function getOrigIdxFromBtn(btn, puzzle) {
  // We store data via data attribute approach — use btn index in grid
  // Instead, re-read from stored data attribute
  return parseInt(btn.dataset.origIdx);
}

function handleAnswer(btn, isCorrect, puzzle) {
  if (answered) return;
  answered = true;
  stopTimer();

  const allBtns = document.querySelectorAll('.shadow-option');
  allBtns.forEach(b => b.classList.add('disabled'));

  const fb = document.getElementById('feedbackBar');

  if (isCorrect) {
    btn.classList.add('correct');
    btn.innerHTML += `<div class="result-badge badge-correct">✓</div>`;

    streak++;
    if (streak > bestStreak) bestStreak = streak;
    const pts = POINTS[puzzle.difficulty] + (streak > 1 ? STREAK_BONUS * (streak - 1) : 0);
    score += pts;
    updateHUD();

    const timeBonus = timeLeft > 0 ? Math.round(timeLeft * 3) : 0;
    score += timeBonus;

    fb.className = 'feedback good show';
    const msgs = [
      `Correct! +${pts} pts${timeBonus ? ` (+${timeBonus} speed bonus)` : ''}`,
      `Sharp eye! +${pts} pts${streak > 2 ? ` 🔥 ${streak}× streak!` : ''}`,
      `Nailed it! +${pts} pts${timeBonus > 20 ? ' ⚡ Fast!' : ''}`
    ];
    fb.textContent = msgs[Math.floor(Math.random() * msgs.length)];

    allBtns.forEach(b => {
      if (!b.classList.contains('correct')) b.classList.add('dimmed');
    });
  } else {
    btn.classList.add('wrong');
    btn.innerHTML += `<div class="result-badge badge-wrong">✗</div>`;
    streak = 0;
    updateHUD();

    allBtns.forEach(b => {
      const svgContent = b.querySelector('svg').innerHTML;
      // Find and mark correct
      const origSvg = puzzle.shadows[puzzle.correctIdx];
      if (svgContent.trim() === origSvg.trim()) {
        b.classList.add('correct');
        b.innerHTML += `<div class="result-badge badge-correct">✓</div>`;
      } else if (!b.classList.contains('wrong')) {
        b.classList.add('dimmed');
      }
    });

    fb.className = 'feedback bad show';
    fb.textContent = `Not quite. Study the ${puzzle.label}'s silhouette carefully — the correct one is highlighted.`;
  }

  showNextBtn();
}

function showNextBtn() {
  setTimeout(() => {
    const nb = document.getElementById('nextBtn');
    nb.className = 'next-btn show';
    nb.textContent = currentRound < puzzles.length - 1 ? 'Next Round →' : 'View Results';
    nb.onclick = advanceRound;
  }, 500);
}

function advanceRound() {
  currentRound++;
  if (currentRound >= puzzles.length) {
    showEnd();
  } else {
    renderRound();
  }
}

function showEnd() {
  stopTimer();
  document.getElementById('gameContent').style.display = 'none';
  const end = document.getElementById('endScreen');
  end.style.display = 'flex';
  end.classList.add('show');

  document.getElementById('progressFill').style.width = '100%';

  const correct = puzzles.filter((_, i) => i < currentRound).length; // approximate
  document.getElementById('finalCorrect').textContent = `${score > 0 ? '✓' : '—'}`;

  // Recount correct from score
  document.getElementById('finalScore').textContent = score;
  document.getElementById('finalStreak').textContent = bestStreak;

  // Count correct answers
  document.getElementById('finalCorrect').textContent = countCorrect;

  const pct = countCorrect / TOTAL_ROUNDS;
  let icon, name, desc;
  if (pct === 1) { icon='🏆'; name='Perfect Observer'; desc='Flawless spatial reasoning. Every shadow matched without hesitation.'; }
  else if (pct >= 0.9) { icon='🎯'; name='Expert Analyst'; desc='Exceptional precision. You read silhouettes with near-perfect accuracy.'; }
  else if (pct >= 0.75) { icon='🧠'; name='Sharp Eye'; desc='Strong visual pattern recognition. Only a few slipped past you.'; }
  else if (pct >= 0.6) { icon='🔍'; name='Keen Observer'; desc='Above average spatial awareness. With practice you can go further.'; }
  else if (pct >= 0.4) { icon='👁'; name='Developing Eye'; desc='You have the instinct — refining your focus will sharpen accuracy.'; }
  else { icon='📐'; name='Trainee'; desc='Shadow reading takes practice. Study the clues carefully next time.'; }

  document.getElementById('ratingIcon').textContent = icon;
  document.getElementById('ratingName').textContent = name;
  document.getElementById('ratingDesc').textContent = desc;
  document.getElementById('endTitle').textContent = name;
  document.getElementById('endSubtitle').textContent = `${countCorrect} / ${TOTAL_ROUNDS} correct · ${score} points`;
}
